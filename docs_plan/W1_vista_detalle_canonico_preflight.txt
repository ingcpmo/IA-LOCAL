def approve_rc(rc_id, body, ...):
    queue = load_review_queue()
    rc = next((r for r in queue if r["rc_id"] == rc_id), None)
    if not rc:
        raise HTTPException(404, "RC not found")
    
    if rc.get("status") in ("approved", "rejected"):
        raise HTTPException(
            status_code=409,
            detail={
                "error": "rc_already_finalized",
                "rc_id": rc_id,
                "current_status": rc["status"],
                "previously_decided_by": rc.get("reviewer","?"),
                "previously_decided_at": rc.get("reviewed_at","?")

EOF
