═══ A2 — CORPUS: ETIQUETAR INTERNAL_SUMMARY ════════════════════════════════

python3 - <<'PY'
import pathlib

HEADER = (
    "[INTERNAL_SUMMARY — NO ES TEXTO REGULATORIO OFICIAL. "
    "Resumen interno de referencia. Fuente autoritativa PENDIENTE. "
    "No citar como texto de CFR/FDA/USP/ICH.]\n\n"
)
AUTORED = {'hplc_data_integrity.txt','hplc_method_validation.txt','fda_oos_guidance_2022.txt'}
DIRS = [
    pathlib.Path('factory/deployments/lab_qc_project/data/regulations'),
    pathlib.Path('factory/workspaces/lab_qc_project/data/regulations'),
]
changed, already, skipped = [], [], []
for base in DIRS:
    if not base.exists(): continue
    for f in base.rglob('*.txt'):
        txt = f.read_text(errors='replace')
        if 'INTERNAL_SUMMARY' in txt[:200]:
            already.append(f.name); continue
        if f.name in AUTORED or '21 CFR' in txt[:400] or 'ICH Q' in txt[:400]:
            f.write_text(HEADER + txt)
            changed.append(str(f))
        else:
            skipped.append(f.name)
print(f'Etiquetados: {len(changed)} | Ya tenían: {len(already)} | Sin cambios: {len(skipped)}')
for c in changed: print(' →', c)
PY

═══ A3 — ENSURE_OLLAMA_ACCESS.SH (idempotente) ═════════════════════════════

# Crear solo si no existe
if [ ! -f factory/scripts/ops/ensure_ollama_access.sh ]; then
cat > factory/scripts/ops/ensure_ollama_access.sh << 'SCRIPT'
#!/usr/bin/env bash
# Garantiza conectividad de un proyecto Docker a Ollama (proceso nativo host).
# Uso: bash ensure_ollama_access.sh <project_id>
set -euo pipefail
PROJECT_ID="${1:-}"; [ -z "$PROJECT_ID" ] && echo "USO: $0 <project_id>" && exit 1

SUBNET=$(docker network ls --format "{{.Name}}" 2>/dev/null | grep "${PROJECT_ID}" | head -1 | \
  xargs -I{} docker network inspect {} 2>/dev/null | \
  python3 -c "import sys,json; d=json.load(sys.stdin); print(d[0]['IPAM']['Config'][0]['Subnet'])" 2>/dev/null || true)

[ -z "$SUBNET" ] && echo "WARN: red Docker de '${PROJECT_ID}' no encontrada" && exit 0
echo "Proyecto: ${PROJECT_ID} | Subred: ${SUBNET}"

if sudo ufw status 2>/dev/null | grep -q "${SUBNET}"; then
  echo "OK: regla UFW ya existe — sin cambios"
else
  sudo ufw allow from "${SUBNET}" to any port 11434 proto tcp comment "Factory:${PROJECT_ID}→Ollama"
  python3 -c "
import sys; sys.path.insert(0,'/home/ing_cpmo')
from factory.core.audit_writer import write_event
write_event('network_access_configured','${PROJECT_ID}',
  {'subnet':'${SUBNET}','port':11434,'source':'ensure_ollama_access.sh'})" 2>/dev/null || true
  echo "Regla añadida."
fi
SCRIPT
chmod +x factory/scripts/ops/ensure_ollama_access.sh
echo "Script creado: factory/scripts/ops/ensure_ollama_access.sh"
else
  echo "OK: ensure_ollama_access.sh ya existe"
fi
