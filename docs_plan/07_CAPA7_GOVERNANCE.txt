═══ 7.2 — DOCUMENTAR re_certification + ROLLBACK (si falta) ════════════════

python3 - <<'PY'
import json, pathlib
rel = pathlib.Path('factory/deployments/lab_qc_project/releases/v0.3.0.json')
if rel.exists():
    r = json.loads(rel.read_text())
    changed = False
    if r.get('release_type') != 're_certification':
        r['release_type'] = 're_certification'; changed = True
    # Nota de rollback explícita
    if 'rollback' not in r:
        r['rollback'] = {
            'strategy': 'redeploy_previous_release',
            'previous_stable': 'v0.2.0',
            'note': ('v0.3.0 es re-certificación: código idéntico a v0.2.0, sin rebuild. '
                     'Rollback = redeploy de v0.2.0 (imagen sha256:dcf81a827689) si una '
                     'regresión apareciera; no hay cambio de imagen que revertir.'),
            'requires_human_approval': True,
        }
        changed = True
    if 'deployment_reconciled' not in r:
        r['deployment_reconciled'] = True
        r['reconcile_note'] = 'running == release v0.3.0 (misma imagen, mismos volúmenes)'
        changed = True
    if changed:
        rel.write_text(json.dumps(r, indent=2, ensure_ascii=False))
        print("releases/v0.3.0.json actualizado (re_certification + rollback + reconcile).")
    else:
        print("OK: releases/v0.3.0.json ya completo.")
else:
    print("releases/v0.3.0.json NO existe — créalo con los datos del deployment activo:")
    print("  status=deployed_and_validated, release_type=re_certification,")
    print("  approval={approved_by:Cesar, decision_origin:human_confirmed},")
    print("  rollback={previous_stable:v0.2.0, requires_human_approval:true}")
PY
