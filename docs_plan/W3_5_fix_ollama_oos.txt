PASO 2 — Reproducir el cuelgue con evidencia (diagnóstico)
------------------------------------------------------------
echo "=== Auth: query SIN api key debe dar 401/403 ==="
curl -s -o /dev/null -w "sin-key HTTP:%{http_code}\n" -m 10 \
  -X POST http://localhost:8102/api/v1/query \
  -H "Content-Type: application/json" \
  -d '{"query":"test"}'

# Descubrir el header que espera el deployment OOS (puede ser X-Api-Key)
GMP_KEY="$(docker exec oos_hplc_investigator_api printenv GMP_API_KEY 2>/dev/null)"
echo "Probando query con timeout de cliente 130s (mistral CPU puede tardar)..."

  -X POST http://localhost:8102/api/v1/query \
  -H "Content-Type: application/json" \
  -H "X-Api-Key: $GMP_KEY" \
  -d '{"query":"Responde solo: OOS OK"}' 2>&1 | tail -20

echo "=== EN PARALELO: logs de Ollama durante la llamada ==="

  || echo "(sin POST /api/generate en aria-ollama)"

echo "=== Logs del contenedor OOS durante la llamada ==="
